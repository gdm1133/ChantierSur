
window.getDevisLines = function() {
    return window.devisLinesState;
};

window.renderV5LotsState = function() {
    const container = document.getElementById('v5-lots-container');
    container.innerHTML = '';
    
    window.devisLinesState.forEach((line, index) => {
        const div = document.createElement('div');
        div.className = 'flex gap-4 items-center bg-white p-3 rounded-xl border border-slate-200 lot-row';
        
        div.innerHTML = \
            <input type="text" class="lot-desc flex-1 bg-slate-50 border border-slate-300 rounded-lg px-3 py-2" placeholder="Description du lot" value="\" oninput="window.updateV5Lot(\, 'designation', this.value)">
            <input type="number" min="0" class="lot-montant w-48 bg-slate-50 border border-slate-300 rounded-lg px-3 py-2" placeholder="Montant FCFA" value="\" oninput="window.updateV5Lot(\, 'montant_indique', this.value)">
            <button type="button" onclick="window.removeV5Lot(\)" class="text-rose-500 font-bold">X</button>
        \;
        container.appendChild(div);
    });
    window.calcV5Totals();
};

window.updateV5Lot = function(index, field, value) {
    if (window.devisLinesState[index]) {
        if (field === 'montant_indique') {
            window.devisLinesState[index][field] = parseFloat(value) || 0;
            // update Q and PU as fallback so PDF has numbers
            window.devisLinesState[index].quantite = 1;
            window.devisLinesState[index].pu_ht = window.devisLinesState[index].montant_indique;
        } else {
            window.devisLinesState[index][field] = value;
        }
        window.calcV5Totals();
    }
};

window.removeV5Lot = function(index) {
    window.devisLinesState.splice(index, 1);
    window.renderV5LotsState();
};

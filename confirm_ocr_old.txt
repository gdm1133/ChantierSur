    lines.forEach((line, index) => {
        const tr = document.createElement('tr');
        tr.className = 'border-b border-slate-200 hover:bg-slate-50 transition-colors bg-white';
        tr.id = 'ocr-val-row-' + index;
        
        // Use regex for numbers formatting
        const pu = line.pu_ht || 0;
        const total = line.montant_indique || (line.quantite * pu) || 0;

        tr.innerHTML = `
            <td class="p-2"><input type="text" class="w-full border border-slate-300 rounded px-2 py-1 text-xs ocr-input-lot" value="${line.lot || ''}"></td>
            <td class="p-2"><input type="text" class="w-full border border-slate-300 rounded px-2 py-1 text-xs ocr-input-des" value="${line.designation || ''}"></td>
            <td class="p-2"><input type="text" class="w-full border border-slate-300 rounded px-2 py-1 text-xs ocr-input-u text-center" value="${line.unite || 'u'}"></td>
            <td class="p-2"><input type="number" class="w-full border border-slate-300 rounded px-2 py-1 text-xs ocr-input-q text-center" value="${line.quantite || 1}"></td>
            <td class="p-2"><input type="number" class="w-full border border-slate-300 rounded px-2 py-1 text-xs ocr-input-pu text-right" value="${pu}"></td>
            <td class="p-2"><input type="number" class="w-full border border-slate-300 rounded px-2 py-1 text-xs ocr-input-total font-bold text-right text-blue-800" value="${total}"></td>
            <td class="p-2 text-center"><button type="button" onclick="document.getElementById('ocr-val-row-${index}').remove()" class="text-rose-500 font-bold p-1 hover:bg-rose-50 rounded">X</button></td>
        `;
        tbody.appendChild(tr);
    });
    
    document.getElementById('ocr-validation-screen').classList.remove('hidden');
    document.getElementById('ocr-upload-section').classList.add('hidden');
};
